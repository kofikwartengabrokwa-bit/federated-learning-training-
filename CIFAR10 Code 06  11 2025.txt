# ============================================================
# NON-IID DATA SPLIT
# ============================================================

def dirichlet_split(
        dataset,
        num_clients=50,
        alpha=0.7
):

    labels = np.array(dataset.targets)

    client_indices = {
        i: [] for i in range(num_clients)
    }

    class_indices = [

        np.where(labels == c)[0]

        for c in range(10)
    ]

    for c in range(10):

        np.random.shuffle(
            class_indices[c]
        )

        proportions = np.random.dirichlet(
            alpha * np.ones(num_clients)
        )

        proportions = (
            np.cumsum(proportions)
            * len(class_indices[c])
        ).astype(int)[:-1]

        splits = np.split(
            class_indices[c],
            proportions
        )

        for i in range(num_clients):

            client_indices[i].extend(
                splits[i]
            )

    return client_indices

client_indices = dirichlet_split(
    train_dataset,
    num_clients=num_clients,
    alpha=0.7
)

# ============================================================
# FeDR CNN MODEL
# ============================================================

class FeDRCNN(nn.Module):

    def __init__(self):

        super().__init__()

        self.features = nn.Sequential(

            nn.Conv2d(
                3,
                64,
                kernel_size=3,
                padding=1
            ),

            nn.BatchNorm2d(64),

            nn.ReLU(),

            nn.Conv2d(
                64,
                64,
                kernel_size=3,
                padding=1
            ),

            nn.BatchNorm2d(64),

            nn.ReLU(),

            nn.MaxPool2d(2),

            nn.Dropout(0.25),

            nn.Conv2d(
                64,
                128,
                kernel_size=3,
                padding=1
            ),

            nn.BatchNorm2d(128),

            nn.ReLU(),

            nn.Conv2d(
                128,
                128,
                kernel_size=3,
                padding=1
            ),

            nn.BatchNorm2d(128),

            nn.ReLU(),

            nn.MaxPool2d(2),

            nn.Dropout(0.25)
        )

        self.classifier = nn.Sequential(

            nn.Linear(
                128 * 8 * 8,
                512
            ),

            nn.ReLU(),

            nn.Dropout(0.5),

            nn.Linear(
                512,
                10
            )
        )

    def forward(self, x):

        x = self.features(x)

        x = x.view(
            x.size(0),
            -1
        )

        x = self.classifier(x)

        return x

# ============================================================
# FEDER REGULARIZATION
# ============================================================

def feder_regularization(
        local_model,
        global_model
):

    reg_loss = 0

    for lp, gp in zip(
            local_model.parameters(),
            global_model.parameters()
    ):

        reg_loss += torch.norm(
            lp - gp,
            p=2
        ) ** 2

    return reg_loss

# ============================================================
# EVALUATION
# ============================================================

def evaluate(
        model,
        loader
):

    model.eval()

    correct = 0
    total = 0

    total_loss = 0

    criterion = nn.CrossEntropyLoss()

    with torch.no_grad():

        for images, labels in loader:

            images = images.to(device)
            labels = labels.to(device)

            outputs = model(images)

            loss = criterion(
                outputs,
                labels
            )

            total_loss += loss.item()

            predictions = torch.argmax(
                outputs,
                dim=1
            )

            correct += (
                predictions == labels
            ).sum().item()

            total += labels.size(0)

    accuracy = correct / total

    avg_loss = total_loss / len(loader)

    return avg_loss, accuracy

test_loader = DataLoader(
    test_dataset,
    batch_size=batch_size,
    shuffle=False,

)

results = {}

# ============================================================
# TRAINING
# ============================================================

for lr in learning_rates:

    print(f"\n===== Learning Rate {lr} =====")

    global_model = FeDRCNN().to(device)

    criterion = nn.CrossEntropyLoss()

    round_losses = []
    round_accuracies = []

    for rnd in range(num_rounds):

        selected_clients = np.random.choice(
            range(num_clients),
            clients_per_round,
            replace=False
        )

        local_weights = []

        # ====================================================
        # CLIENT TRAINING
        # ====================================================

        for client_id in selected_clients:

            indices = client_indices[client_id]

            if len(indices) < batch_size:
                continue

            subset = Subset(
                train_dataset,
                indices
            )

            loader = DataLoader(
                subset,
                batch_size=batch_size,
                shuffle=True,
                num_workers=2,
                pin_memory=True
            )

            local_model = copy.deepcopy(
                global_model
            ).to(device)

            optimizer = optim.AdamW(
                local_model.parameters(),
                lr=lr,
                weight_decay=1e-4
            )

            scheduler = optim.lr_scheduler.CosineAnnealingLR(
                optimizer,
                T_max=local_epochs
            )

            local_model.train()

            # ====================================================
            # LOCAL EPOCHS
            # ====================================================

            for epoch in range(local_epochs):

                for images, labels in loader:

                    images = images.to(device)
                    labels = labels.to(device)

                    optimizer.zero_grad()

                    outputs = local_model(images)

                    ce_loss = criterion(
                        outputs,
                        labels
                    )

                    reg_loss = feder_regularization(
                        local_model,
                        global_model
                    )

                    loss = (
                        ce_loss
                        + fed_lambda * reg_loss
                    )

                    loss.backward()

                    torch.nn.utils.clip_grad_norm_(
                        local_model.parameters(),
                        5.0
                    )

                    optimizer.step()

                scheduler.step()

            local_weights.append(
                copy.deepcopy(
                    local_model.state_dict()
                )
            )

        # ====================================================
        # FEDERATED AVERAGING
        # ====================================================

        global_weights = global_model.state_dict()

        for key in global_weights.keys():

            global_weights[key] = torch.stack([

                local_weights[i][key].float()

                for i in range(len(local_weights))

            ], dim=0).mean(dim=0)

        global_model.load_state_dict(
            global_weights
        )

        # ====================================================
        # EVALUATION
        # ====================================================

        test_loss, test_acc = evaluate(
            global_model,
            test_loader
        )

        round_losses.append(
            test_loss
        )

        round_accuracies.append(
            test_acc
        )

        if (rnd + 1) % 10 == 0:

            print(
                f"Round [{rnd+1}/{num_rounds}] | "
                f"Loss: {test_loss:.4f} | "
                f"Accuracy: {test_acc:.4f}"
            )

    results[str(lr)] = {

        "loss": round_losses,
        "accuracy": round_accuracies
    }

# SMOOTH FUNCTION
# ============================================================

def smooth(signal, alpha=0.2):

    smoothed = []

    last = signal[0]

    for point in signal:

        last = (
            alpha * point
            + (1 - alpha) * last
        )

        smoothed.append(last)

    return smoothed
