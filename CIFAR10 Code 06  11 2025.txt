import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader, random_split
import copy
import numpy as np
import matplotlib.pyplot as plt
import os

# --- CIFAR-10 Dataset ---
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))
])

train_dataset = torchvision.datasets.CIFAR10(root='./data', train=True, download=True, transform=transform)
test_dataset = torchvision.datasets.CIFAR10(root='./data', train=False, download=True, transform=transform)

num_clients = 1
client_datasets = random_split(train_dataset, [len(train_dataset)//num_clients]*num_clients)

# --- CNN Actor-Critic Model ---
class CNNActorCritic(nn.Module):
    def __init__(self, num_actions=10):
        super(CNNActorCritic, self).__init__()
        self.conv1 = nn.Conv2d(3, 32, 3, padding=1)
        self.conv2 = nn.Conv2d(32, 64, 3, padding=1)
        self.pool = nn.MaxPool2d(2, 2)
        self.fc = nn.Linear(64*8*8, 128)
        self.actor = nn.Linear(128, num_actions)
        self.critic = nn.Linear(128, 1)

    def forward(self, x):
        x = F.relu(self.conv1(x))
        x = self.pool(x)
        x = F.relu(self.conv2(x))
        x = self.pool(x)
        x = x.view(x.size(0), -1)
        x = F.relu(self.fc(x))
        return F.softmax(self.actor(x), dim=-1), self.critic(x)

# --- Local training function ---
def train_local(model, optimizer, data_loader, device):
    model.train()
    total_loss, total_reward, correct, total = 0, 0, 0, 0
    criterion = nn.CrossEntropyLoss()

    for images, labels in data_loader:
        images, labels = images.to(device), labels.to(device)
        probs, value = model(images)
        dist = torch.distributions.Categorical(probs)
        actions = labels  # supervised labels as actions
        log_probs = dist.log_prob(actions)
        reward = (actions == actions).float()  # always 1
        advantage = reward - value.squeeze()

        actor_loss = -(log_probs * advantage.detach()).mean()
        critic_loss = advantage.pow(2).mean()
        loss = actor_loss + critic_loss

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        total_loss += loss.item()
        total_reward += reward.mean().item()
        correct += (probs.argmax(dim=1) == labels).sum().item()
        total += labels.size(0)

    accuracy = 1000*correct/total
    return copy.deepcopy(model.state_dict()), total_reward / len(data_loader), total_loss / len(data_loader), accuracy


def dr_weights(local_weights, rewards):
    best_idx = np.argmax(rewards)
    return local_weights[best_idx]

# --- Federated learning ---
def federated_learning(
    algorithm='L2FPPO',
    num_servers=1,
    num_clients=1,
    num_rounds=31,
    lambda_lr=0.01,
    queue_step=0.01
):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    clients = [CNNActorCritic().to(device) for _ in range(num_clients)]
    optimizers = [optim.Adam(c.parameters(), lr=1e-5) for c in clients]
    global_model = CNNActorCritic().to(device)

    reward_curve, loss_curve, acc_curve = [], [], []
    lambdas = [0.1 for _ in range(num_clients)]
    queue_lengths = [1.0 for _ in range(num_clients)]
    powers = [1.0 for _ in range(num_clients)]

    for rnd in range(num_rounds):
        local_weights, local_rewards, local_losses, local_accs = [], [], [], []

        for i, client in enumerate(clients):
            loader = DataLoader(client_datasets[i], batch_size=32, shuffle=True)
            client.train()
            total_loss, total_reward, correct, total = 0, 0, 0, 0

            for images, labels in loader:
                images, labels = images.to(device), labels.to(device)
                probs, value = client(images)
                dist = torch.distributions.Categorical(probs)
                actions = labels
                log_probs = dist.log_prob(actions)
                reward = (probs.argmax(dim=1) == labels).float().mean()  # accuracy as reward proxy

                advantage = reward - value.mean()
                actor_loss = -(log_probs * advantage.detach()).mean()
                critic_loss = advantage.pow(2).mean()

                # --- Lagrange–Lyapunov loss integration ---
                if algorithm == 'L2FPPO':
                    lagrangian_term = lambdas[i] * (queue_lengths[i] - powers[i])
                    lyapunov_term = 0.5 * (queue_lengths[i] ** 2)
                    loss = actor_loss + critic_loss + 0.01 * lagrangian_term + 0.001 * lyapunov_term
                else:
                    loss = actor_loss + critic_loss

                optimizers[i].zero_grad()
                loss.backward()
                optimizers[i].step()

                total_loss += loss.item()
                total_reward += reward.item()
                correct += (probs.argmax(dim=1) == labels).sum().item()
                total += labels.size(0)

            avg_acc = correct / total
            avg_reward = total_reward / len(loader)
            avg_loss = total_loss / len(loader)

            # Update Lyapunov queue and Lagrange multiplier
            queue_lengths[i] += queue_step * (queue_lengths[i] - avg_reward)
            lambdas[i] = max(0, lambdas[i] + lambda_lr * (queue_lengths[i] - powers[i]))

            local_weights.append(copy.deepcopy(client.state_dict()))
            local_rewards.append(avg_reward)
            local_losses.append(avg_loss)
            local_accs.append(avg_acc)

            # Print per-client details
            print(f"Client {i+1}: λ={lambdas[i]:.4f}, Q={queue_lengths[i]:.4f}, "
                  f"Reward={avg_reward:.4f}, Loss={avg_loss:.4f}, Acc={avg_acc:.4f}")

        # --- Aggregation phase ---
        if algorithm == "L2FPPO":
            new_weights =dr_weights(local_weights, local_rewards)       
        else:
            raise ValueError("Unknown algorithm")

        global_model.load_state_dict(new_weights)
        for client in clients:
            client.load_state_dict(new_weights)

        # --- Global logging ---
        reward_curve.append(np.mean(local_rewards))
        loss_curve.append(np.mean(local_losses))
        acc_curve.append(np.mean(local_accs))

        print(f"[{algorithm}] Round {rnd+1}/{num_rounds} | "
              f"Avg Reward={reward_curve[-1]:.4f} | Avg Loss={loss_curve[-1]:.4f} | "
              f"Avg Acc={acc_curve[-1]:.4f}\n")

    return reward_curve, loss_curve, acc_curve


# --- Run all four algorithms ---
algorithms = ["L2FPPO"]
results = {}

for algo in algorithms:
    print(f"=== Training {algo} ===")
    reward, loss, acc = federated_learning(algo, num_rounds=31)
    results[algo] = {"reward": reward, "loss": loss, "acc": acc}
